    renderSection1(state),
    renderSection2(state),
    renderSection3(state),
    renderSection4(state),
    renderSection5(state),
    renderSection6(state),
    renderSection7(state),

    renderSection9(state),
    renderSection10(state),
    renderSection11(state),
    renderSection12(state),
    renderSection13(state),
    renderSection14(state),
    renderSection15(state),
    renderSection16(state),
    renderSection17(state),
    renderSection18(state),
  ];
  return {
    status: statusLabel(state?.status),
    sections,
    total: sections.reduce((a, s) => a + s.metric, 0),
  };
